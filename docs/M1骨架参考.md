```ts
/** 归一化动作：listener 把 session 事件翻译成计分引擎可消费的最小结构 */
export type Action =
  | { kind: 'turn-completed'; turn: number }
  | { kind: 'turn-failed'; turn: number }
  | { kind: 'turn-aborted'; turn: number }
  | { kind: 'tool-call'; tool: string }
  | { kind: 'tool-failed'; tool: string }
  | { kind: 'todo-completed'; count: number }
  | { kind: 'tokens'; tokens: number }
  | { kind: 'subagent'; depth: number }
  | { kind: 'session-start'; hourOfDay: number; source: string }

/** 计数器（成就判定的输入） */
export interface Counters {
  turnsCompleted: number
  turnsFailed: number
  consecutiveSuccess: number
  toolCalls: number
  toolCallsByTool: Record<string, number>
  todosCompleted: number
  tokensOut: number
  subagentsSpawned: number
  activeDays: number
  lastActiveDay: string      // 'YYYY-MM-DD'
  lastActivityAt: number     // epoch ms
}

/** 存档 */
export interface SaveData {
  version: number
  cwd: string
  level: number
  xp: number
  xpTotal: number
  title: string
  season: string
  counters: Counters
  achievements: Record<string, { acquiredAt: number; xp: number }>
  lastSeqBySession: Record<string, number>   // 幂等水位
  updatedAt: number
}

/** 成就定义 */
export interface AchievementDef {
  id: string
  name: { zh: string; en: string }
  description: { zh: string; en: string }
  icon: string
  xp: number
  hidden?: boolean
  check: (c: Counters) => boolean
}
```

## 4. src/engine.ts（纯函数，全部可单测）

```ts
import type { Action, Counters, SaveData } from './types.ts'

export const TITLES = [
  { min: 1, zh: '学徒', en: 'Apprentice' },
  { min: 5, zh: '工匠', en: 'Artisan' },
  { min: 10, zh: '锻造师', en: 'Forger' },
  { min: 15, zh: '宗师', en: 'Master' },
  { min: 20, zh: '传说', en: 'Legend' },
]

export function xpToNext(level: number): number {
  return Math.round(100 * Math.pow(level, 1.5))
}

export function titleFor(level: number): { zh: string; en: string } {
  let t = TITLES[0]
  for (const cand of TITLES) if (level >= cand.min) t = cand
  return { zh: t.zh, en: t.en }
}

/** 工具 XP：锻造师工具 +2，其余 +1；返回 null 表示不奖励 */
const CRAFT_TOOLS = new Set(['edit', 'write', 'str-replace-editor', 'pwsh', 'bash', 'ssh_exec'])

function xpForTool(tool: string): number {
  return CRAFT_TOOLS.has(tool) ? 2 : 1
}

/** 核心计分：Action → 增加的 XP（连击加成在 turn 级结算，见 applyTurn） */
export function xpForAction(action: Action): number {
  switch (action.kind) {
    case 'turn-completed': return 10
    case 'turn-failed': return 2
    case 'tool-call': return xpForTool(action.tool)
    case 'todo-completed': return 15 * action.count
    case 'tokens': return Math.floor(action.tokens / 10_000)
    default: return 0
  }
}

/** 单回合结算：聚合该回合的工具调用，应用连击加成 */
export function applyTurn(save: SaveData, actions: Action[]): SaveData {
  const s = structuredClone(save)
  const c = s.counters
  let gain = 0

  for (const a of actions) {
    gain += xpForAction(a)
    if (a.kind === 'tool-call') { c.toolCalls++; c.toolCallsByTool[a.tool] = (c.toolCallsByTool[a.tool] ?? 0) + 1 }
    if (a.kind === 'todo-completed') c.todosCompleted += a.count
    if (a.kind === 'tokens') c.tokensOut += a.tokens
  }

  const completed = actions.some(a => a.kind === 'turn-completed')
  const failed = actions.some(a => a.kind === 'turn-failed')

  if (completed) {
    c.turnsCompleted++
    c.consecutiveSuccess++
    if (c.consecutiveSuccess >= 5) gain = Math.round(gain * 1.5)   // 连击 ×1.5
  } else if (failed) {
    c.turnsFailed++
    c.consecutiveSuccess = 0
  }

  gain = Math.min(gain, 10 + 15 + 100)   // 单回合兜底上限（工具 10 + todo 15 + turn 基础，宽松）
  return addXp(s, gain)
}

/** 加 XP 并处理升级（返回原 save 副本；升级信息由调用方从 level 变化推断） */
export function addXp(save: SaveData, gain: number): SaveData {
  const s = structuredClone(save)
  s.xp += gain
  s.xpTotal += gain
  s.counters.lastActivityAt = Date.now()
  while (s.xp >= xpToNext(s.level)) {
    s.xp -= xpToNext(s.level)
    s.level++
  }
  s.title = titleFor(s.level).zh
  s.updatedAt = Date.now()
  return s
}

/** 成就判定：返回新解锁的成就 id 列表（一次性） */
export function checkAchievements(defs: AchievementDef[], save: SaveData): string[] {
  const unlocked: string[] = []
  for (const d of defs) {
    if (save.achievements[d.id]) continue           // 一次性
    if (d.check(save.counters)) {
      save.achievements[d.id] = { acquiredAt: Date.now(), xp: d.xp }
      unlocked.push(d.id)
    }
  }
  return unlocked
}
```

## 5. src/listener.ts（事件订阅 → Action 归一化）

```ts
import type { Context } from '@deepseek-ai/cordis'
import type { SessionEvent } from '@deepseek-ai/dsh-session'
import type { Action } from './types.ts'

/** 订阅 session/event，按 session.id + event.seq 幂等去重，输出 Action 流 */
export function watchEvents(ctx: Context, emit: (sessionId: string, seq: number, action: Action) => void): void {
  const seen = new Set<string>()

  ctx.on('session/event', (session, event) => {
    const key = `${session.id}:${event.seq}`
    if (seen.has(key)) return
    seen.add(key)

    try {
      const a = normalize(event)
      if (a) emit(String(session.id), event.seq, a)
    } catch (err) {
      // 监听器绝不抛出：只记录，不影响 session 提交
      console.error('[devquest] event error:', err)
    }
  })
}

function normalize(event: SessionEvent): Action | null {
  switch (event.type) {
    case 'turn/end':
      if (event.data.reason.kind === 'completed') return { kind: 'turn-completed', turn: event.data.turn }
      if (event.data.reason.kind === 'error') return { kind: 'turn-failed', turn: event.data.turn }
      if (event.data.reason.kind === 'aborted') return { kind: 'turn-aborted', turn: event.data.turn }
      return null
    case 'tool/call':
      return { kind: 'tool-call', tool: event.data.name }
    case 'tool/result':
      if (event.data.error) return { kind: 'tool-failed', tool: '?' }   // 需与 tool/call 配对取 name（M2 完善）
      return null
    case 'todo/write': {
      const newly = event.data.todos.filter(t => t.status === 'completed').length
      return newly > 0 ? { kind: 'todo-completed', count: newly } : null
    }
    case 'assistant/message':
      return event.data.usage?.totalTokens
        ? { kind: 'tokens', tokens: event.data.usage.totalTokens }
        : null
    case 'request/context':
      // subagent 判定需结合 session.header.delegationDepth，见 index.ts 组装
      return null
    default:
      return null
  }
}
```

## 6. src/index.ts（装配）

```ts
import type { Context } from '@deepseek-ai/cordis'
import { defineTool, type JsonValue } from '@deepseek-ai/dsh-tools'
import type {} from '@deepseek-ai/dsh-fs'
import type {} from '@deepseek-ai/dsh-session'
import { watchEvents } from './listener.ts'
import { addXp, applyTurn, checkAchievements, titleFor } from './engine.ts'
import { loadSave, persistSave } from './store.ts'
import { ACHIEVEMENTS } from './achievements.ts'

export const name = 'devquest'
export const inject = ['fs', 'sessions', 'tools'] as const

export interface Config {
  /** 存档根目录（缺省 ~/.dsh/devquest） */
  dataDir?: string
  /** 赛季名 */
  season?: string
}

export function apply(ctx: Context, config: Config = {}): void {
  // 1. 事件订阅：按会话聚合 Action，回合结束结算
  const turnBuffer = new Map<string, unknown[]>()   // sessionId → actions（M2 完善类型）

  watchEvents(ctx, (sessionId, seq, action) => {
    const buf = turnBuffer.get(sessionId) ?? []
    buf.push(action)
    if (action.kind === 'turn-completed' || action.kind === 'turn-failed') {
      const save = loadSave(ctx, sessionId, config)
      const next = applyTurn(save, buf as never)
      const unlocked = checkAchievements(ACHIEVEMENTS, next)
      persistSave(ctx, sessionId, next)
      turnBuffer.delete(sessionId)
      if (unlocked.length > 0) console.log(`[devquest] 🏆 ${unlocked.join(', ')}`)
    } else {
      turnBuffer.set(sessionId, buf)
    }
  })

  // 2. 工具：devquest_status
  ctx.tools.register(defineTool({
    name: 'devquest_status',
    description: '查询 DevQuest 开发游戏化进度（等级/XP/成就）。',
    parameters: {},
    output: { schema: { type: 'object', additionalProperties: true } },
    async execute(): Promise<Record<string, JsonValue>> {
      const save = loadSave(ctx, '<current>', config)
      return {
        level: save.level,
        xp: save.xp,
        xpToNext: 100 * Math.round(Math.pow(save.level, 1.5)),
        title: save.title,
        counters: save.counters,
        achievements: Object.keys(save.achievements),
      } as Record<string, JsonValue>
    },
  }))
}

// store.ts 与 achievements.ts 见开发文档 §4/§6，此处略（结构已定，实现按文档）
```

## 7. tests/engine.test.ts（单测骨架）

```ts
import test from 'node:test'
import assert from 'node:assert/strict'
import { applyTurn, xpToNext, titleFor, checkAchievements } from '../src/engine.ts'

test('xpToNext 等级曲线', () => {
  assert.equal(xpToNext(1), 100)
  assert.equal(xpToNext(10), 3074)   // round(100 × 10^1.5)
})

test('completed turn 加 10 XP', () => {
  const save = freshSave()
  const next = applyTurn(save, [{ kind: 'turn-completed', turn: 1 }])
  assert.equal(next.xp, 10)
})

test('连击 5 轮后 ×1.5', () => {
  let save = freshSave()
  for (let i = 0; i < 5; i++) save = applyTurn(save, [{ kind: 'turn-completed', turn: i + 1 }])
  assert.equal(save.counters.consecutiveSuccess, 5)
  assert.equal(save.counters.turnsCompleted, 5)
})

test('error 清零连击', () => {
  let save = freshSave()
  save = applyTurn(save, [{ kind: 'turn-completed', turn: 1 }])
  save = applyTurn(save, [{ kind: 'turn-failed', turn: 2 }])
  assert.equal(save.counters.consecutiveSuccess, 0)
})

test('成就一次性', () => {
  let save = freshSave()
  save.counters.turnsCompleted = 10
  const first = checkAchievements([ACH('turns_10')], save)
  assert.deepEqual(first, ['turns_10'])
  const again = checkAchievements([ACH('turns_10')], save)
  assert.deepEqual(again, [])
})

function freshSave() { /* 构造最小 SaveData */ }
function ACH(id: string) { return ACHIEVEMENTS.find(a => a.id === id)! }
```
